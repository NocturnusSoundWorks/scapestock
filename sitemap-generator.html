<script>


function generateSitemap(){


let base="https://nocturnussoundworks.github.io/scapestock/"


let xml=`<?xml version="1.0" encoding="UTF-8"?>

<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">

`


/* トップページ */


xml+=`

<url>

<loc>${base}</loc>

</url>

`


/* タグページ */


const tags=[

"japan",

"traditional",

"sakura",

"nature",

"sky",

"flower",

"night",

"rain",

"city",

"texture",

"scenery"

]


tags.forEach(tag=>{


xml+=`

<url>

<loc>${base}index.html?tag=${tag}</loc>

</url>

`


})


/* 写真ページ */


PHOTOS.forEach(photo=>{


xml+=`

<url>

<loc>${base}photo.html?id=${photo.id}</loc>

</url>

`


})


xml+=`</urlset>`


/* ファイルダウンロード */


const blob=new Blob([xml],{type:"text/xml"})

const url=URL.createObjectURL(blob)


const a=document.createElement("a")

a.href=url

a.download="sitemap.xml"

a.click()


}
